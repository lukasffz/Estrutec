USE estrutec;

-- TABELAS
CREATE TABLE cadastrados (
    id_login      INT PRIMARY KEY AUTO_INCREMENT,
    nome          VARCHAR(150) NOT NULL,
    email         VARCHAR(100) UNIQUE NOT NULL,
    senha         VARCHAR(255) NOT NULL,
    telefone      VARCHAR(20) NOT NULL,
    cpf           VARCHAR(20) NOT NULL,
    papel         VARCHAR(20) DEFAULT 'cliente',
    ativo         TINYINT(1) DEFAULT 1,
    online        TINYINT(1) DEFAULT 0
);
CREATE TABLE enderecos (
    id_endereco INT PRIMARY KEY AUTO_INCREMENT,
    id_login    INT NOT NULL,
    rua         VARCHAR(150) NOT NULL,
    numero      VARCHAR(20) NOT NULL,
    bairro      VARCHAR(100) NOT NULL,
    cidade      VARCHAR(100) NOT NULL,
    cep         VARCHAR(20) NOT NULL,
    FOREIGN KEY (id_login) REFERENCES cadastrados(id_login) ON DELETE CASCADE
);
CREATE TABLE produtos (
    id          INT PRIMARY KEY AUTO_INCREMENT,
    item        VARCHAR(100) NOT NULL,
    categoria   VARCHAR(100) NOT NULL,
    descricao   VARCHAR(300),
    quantidade  INT UNSIGNED DEFAULT 0, -- Alterado para UNSIGNED (Apenas positivos e zero)
    imagem      VARCHAR(200),
    preco       DECIMAL(8,2) NOT NULL,
    status      TINYINT(1) DEFAULT 1    -- ADICIONADO AQUI: 1 para Ativo, 0 para Inativo
);
CREATE TABLE pedidos (
    id_pedido   INT PRIMARY KEY AUTO_INCREMENT,
    id_login    INT NOT NULL,
    data_pedido DATETIME DEFAULT CURRENT_TIMESTAMP,
    status      VARCHAR(50) DEFAULT 'Pendente',
    total       DECIMAL(8,2) NOT NULL,
    FOREIGN KEY (id_login) REFERENCES cadastrados(id_login)
);
CREATE TABLE itens_pedido (
    id_item        INT PRIMARY KEY AUTO_INCREMENT,
    id_pedido      INT NOT NULL,
    id_produto     INT NOT NULL,
    qtd_comprada   INT NOT NULL,
    preco_unitario DECIMAL(8,2) NOT NULL,
    FOREIGN KEY (id_pedido)  REFERENCES pedidos(id_pedido) ON DELETE CASCADE,
    FOREIGN KEY (id_produto) REFERENCES produtos(id)
);
